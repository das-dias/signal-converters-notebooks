    Problem 9¶

    A1 = 100e-3
f1 = 1e3
ph1 = 0
s1 = A1 * np.sin(2 * np.pi * f1 * tD + ph1)

vref = 5  # [V]
nbit = 12
vlsb = vref / (2**nbit)

p_nq = (vlsb**2) / 12  # [V^2]
v_nq = np.sqrt(p_nq)  # [V]

pin = (A1**2) / 2  # [V^2]

snr = 10 * np.log10(pin / p_nq)  # [dB]
print("SNR for A_in = 100 mV : {:.2f} dB".format(snr))

    SNR for A_in = 100 mV : 46.05 dB


    A2 = vref / 2
snr_max = 10 * np.log10((A2**2 / 2) / p_nq)  # [dB]
snr_max_compare = 6.02 * nbit + 1.76  # [dB]
print("Maximum SNR for A_in = V_ref/2 : {:.2f} dB".format(snr_max))
print("Maximum SNR (ideal expression) : {:.2f} dB".format(snr_max_compare))
assert np.isclose(snr_max, snr_max_compare, 1e-2)

    Maximum SNR for A_in = V_ref/2 : 74.01 dB
Maximum SNR (ideal expression) : 74.00 dB


    s2 = A2 * np.sin(2 * np.pi * f1 * tD + ph1) + vref / 2

    Problem 10¶

    vref = 4  # [V]
nbit = 3
vlsb = vref / (2**nbit)
din = np.arange(0, 2**nbit, 1).astype(int)

vout_nonideal = np.array([0.011, 0.507, 1.002, 1.501, 1.996, 2.495, 2.996, 3.491])
vout_ideal = din * vlsb

fig, ax = plt.subplots(1, 1, figsize=(3, 3))
ax.step(
    din, vout_nonideal, color="b", linewidth=2, marker="o", markersize=2, mfc="None"
)
ax.set_xlabel("Input Code")
ax.set_ylabel("Output Voltage (V)")
ax.set_title("Output characteristic of the DAC")
ax.grid()
plt.show()

    offset = (vout_nonideal[0] - vout_ideal[0]) / vlsb  # [LSB]
print("Offset error       : {:.2f} LSB".format(offset))

end_of_scale = (vout_nonideal[-1] - vout_ideal[-1]) / vlsb  # [LSB]
print("End of scale error : {:.2f} LSB".format(end_of_scale))

    Offset error       : 0.02 LSB
End of scale error : -0.02 LSB

    The expressions for DNL and INL already remove the offset and gain error of the DAC through the computation of the measured LSB voltage.

    fig, ax = plt.subplots(1, 1, figsize=(4, 4))
ax.step(
    din,
    vout_nonideal,
    color="r",
    linewidth=2,
    marker="o",
    markersize=2,
    mfc="None",
    label="Non-ideal",
)
ax.step(
    din,
    vout_ideal,
    color="k",
    linewidth=2,
    marker="o",
    markersize=2,
    mfc="None",
    label="Ideal",
)
ax.set_xlabel("Input Code")
ax.set_ylabel("Output Voltage (V)")
ax.set_title("Output characteristic of the DAC")
ax.legend()
ax.grid(True)
plt.show()

    vlsb_measured = (vout_nonideal[-1] - vout_nonideal[0]) / ((2**nbit) - 1)
print("Measured LSB    : {:.2f} mV".format(vlsb_measured * 1e3))
print("Theoretical LSB : {:.2f} mV".format(vlsb * 1e3))

dnl = np.diff(vout_nonideal) / vlsb_measured - 1
dnl_classic = (vout_nonideal[1:] - vout_nonideal[:-1]) / vlsb_measured - 1
assert np.allclose(dnl, dnl_classic)

inl = (vout_nonideal - vlsb_measured * din - vout_nonideal[0]) / vlsb_measured
inl_cumsum = np.array([0] + np.cumsum(dnl).tolist())
dnl_diff = np.diff(inl)

fig, ax = plt.subplots(2, 1, figsize=(4, 4))
ax[0].plot(din[1:], dnl, color="b", linewidth=2, marker="o", markersize=5, mfc="None")
ax[0].plot(
    din[1:], dnl_diff, color="k", linewidth=2, marker="o", markersize=5, mfc="None"
)
ax[0].set_ylabel("DNL")
ax[0].set_xticks(din)
ax[1].plot(din, inl, color="b", linewidth=2, marker="o", markersize=5, mfc="None")
ax[1].plot(
    din, inl_cumsum, color="k", linewidth=2, marker="o", markersize=5, mfc="None"
)
ax[1].set_xlabel("Input Code")
ax[1].set_xticks(din)
ax[1].set_ylabel("INL")
ax[0].grid(True)
ax[1].grid(True)
plt.show()

    Measured LSB    : 497.14 mV
Theoretical LSB : 500.00 mV

    Note: The first and last points of the INL are always 0.0 LSB. This is guaranteed because the expression used to compute the INL automatically removes the offset and gain error of the DAC (by subtracting vlsb_measured * din[0] and dividing by vlsb_measured, respectively).

    dac_linearity = nbit - np.log2(np.max(inl) - np.min(inl))
print("Effective number of bits (ENOB) : {:.2f}".format(dac_linearity))

    Effective number of bits (ENOB) : 9.44

    The linearity of the DAC does not measure its resolution. It measures how closely each code transition aligns with those of an ideal DAC with 9.44 bits of resolution. As max(INL) - min(INL) converges toward 0 LSB, the linearity converges toward infinity, meaning the DAC can preserve resolution equivalent to an infinite number of bits during digital-to-analog conversion.

    fbin = fD[np.argmin(np.abs(fD - f3))]
assert not np.all(fbin == np.nan)
dout_p_fin = dout_p[fD == fbin][0]

new_zero = 1e-10

    vni = 0.1e-3
noise_step = (vlsb / 2 - vni) / 10
noise = np.arange(vni, vlsb / 2, noise_step)
qnoise = []
snrs = []

for vn in noise:
    s3 = A3 * np.sin(2 * np.pi * f3 * tD + ph1) + dc_level + normal(0, vn, Np)
    dout = np.round(s3 / vlsb).astype(int)
    dout_f = np.abs(np.fft.fftshift(np.fft.fft(dout) / Np))
    dout_p = dout_f * dout_f
    dout_p[fD == 0] = new_zero
    fbin = fD[np.argmin(np.abs(fD - f3))]
    dout_p_fin = dout_p[fD == fbin]
    ideal_spectrum = new_zero * np.ones(len(dout_p))
    ideal_spectrum[fD == fbin] = dout_p_fin
    noise_spectrum = dout_p - ideal_spectrum
    noise_spectrum[noise_spectrum == 0] = new_zero
    q_noise_p = np.mean(noise_spectrum)
    snr = 10 * np.log10(dout_p_fin / q_noise_p)  # [dB]
    qnoise.append(vn**2)
    snrs.append(snr)

fig, ax = plt.subplots(1, 1, figsize=(4, 4))
ax.plot(np.array(qnoise), np.array(snrs), marker="o", color="b")
ax.set_xlabel("Quantization noise ($V^2$)")
ax.set_ylabel("SNR (dB)")
ax.set_title("SNR vs. Quantization noise")
ax.grid(True)
plt.show()
